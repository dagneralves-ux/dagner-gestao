Configuração inicial — só uma vez
⚡ Conectar ao Supabase
Banco de dados gratuito online. Leva ~5 min de configuração.
Primeira vez? Ver como criar o banco
1
Acesse supabase.com → Start your project → crie conta gratuita.
2
Clique em New project, dê um nome e aguarde ~2 min.
3
SQL Editor → New query → cole e execute:
-- ── Tabela clientes ──
CREATE TABLE clientes (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id uuid REFERENCES auth.users(id) ON DELETE CASCADE,
  empresa text NOT NULL,
  responsavel text NOT NULL,
  telefone text DEFAULT '',
  qtd_midias int DEFAULT 0,
  dias int[] DEFAULT '{}',
  observacoes text DEFAULT '',
  logo_url text DEFAULT '',
  status text DEFAULT 'ativo',
  instagram text DEFAULT '',
  valor_mensal numeric(10,2) DEFAULT 0,
  dia_cobranca int DEFAULT 0,
  criado_em timestamptz DEFAULT now()
);
ALTER TABLE clientes ENABLE ROW LEVEL SECURITY;
CREATE POLICY "user_owns_clientes" ON clientes
  FOR ALL USING (auth.uid() = user_id)
  WITH CHECK (auth.uid() = user_id);

-- ── Tabela entregas ──
CREATE TABLE entregas (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id uuid REFERENCES auth.users(id) ON DELETE CASCADE,
  cliente_id uuid REFERENCES clientes(id) ON DELETE CASCADE,
  data_entrega date NOT NULL,
  criado_em timestamptz DEFAULT now()
);
ALTER TABLE entregas ENABLE ROW LEVEL SECURITY;
CREATE POLICY "user_owns_entregas" ON entregas
  FOR ALL USING (auth.uid() = user_id)
  WITH CHECK (auth.uid() = user_id);

-- ── Tabela lembretes (anotações do calendário) ──
CREATE TABLE lembretes (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id uuid REFERENCES auth.users(id) ON DELETE CASCADE,
  data_lembrete date NOT NULL,
  texto text NOT NULL,
  criado_em timestamptz DEFAULT now()
);
ALTER TABLE lembretes ENABLE ROW LEVEL SECURITY;
CREATE POLICY "user_owns_lembretes" ON lembretes
  FOR ALL USING (auth.uid() = user_id)
  WITH CHECK (auth.uid() = user_id);

-- ── Tabela tarefas (tarefas do dia) ──
CREATE TABLE tarefas (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id uuid REFERENCES auth.users(id) ON DELETE CASCADE,
  data_tarefa date NOT NULL,
  texto text NOT NULL,
  cliente_id uuid REFERENCES clientes(id) ON DELETE SET NULL,
  concluida boolean DEFAULT false,
  criado_em timestamptz DEFAULT now()
);
ALTER TABLE tarefas ENABLE ROW LEVEL SECURITY;
CREATE POLICY "user_owns_tarefas" ON tarefas
  FOR ALL USING (auth.uid() = user_id)
  WITH CHECK (auth.uid() = user_id);

-- ── Buckets de armazenamento (execute no Storage) ──
-- Crie dois buckets públicos no painel Storage do Supabase:
--   "logos"  → para logos dos clientes
--   "brand"  → para a logo do seu estúdio

-- ── Tabela pagamentos ──
CREATE TABLE pagamentos (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id uuid REFERENCES auth.users(id) ON DELETE CASCADE,
  cliente_id uuid REFERENCES clientes(id) ON DELETE CASCADE,
  mes_referencia text NOT NULL,
  valor_pago numeric(10,2) NOT NULL DEFAULT 0,
  data_pagamento timestamptz DEFAULT now(),
  forma_pagamento text DEFAULT 'pix',
  observacao text DEFAULT '',
  criado_em timestamptz DEFAULT now(),
  UNIQUE(user_id, cliente_id, mes_referencia)
);
ALTER TABLE pagamentos ENABLE ROW LEVEL SECURITY;
CREATE POLICY "user_owns_pagamentos" ON pagamentos
  FOR ALL USING (auth.uid() = user_id)
  WITH CHECK (auth.uid() = user_id);
